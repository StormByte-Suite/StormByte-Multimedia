function(add_pipeline_test test_name)
	add_test(NAME "pipeline.${test_name}" COMMAND MultimediaPipelineTests "${test_name}")
	set_tests_properties("pipeline.${test_name}" PROPERTIES TIMEOUT 30 LABELS "multimedia;pipeline")
endfunction()

function(add_optional_pipeline_test test_name)
	add_pipeline_test(${test_name})
	set_tests_properties("pipeline.${test_name}" PROPERTIES SKIP_RETURN_CODE 77)
endfunction()
